    () =>

        }
        return (
          (!item.teacherOnly || isTeacher) && (!item.adminOnly || isAdmin)
        );
      }),

    const closeMenus = () => {

              className={`text-indigo-200 hover:text-white hover:bg-white/10 p-2 rounded-lg transition ${
